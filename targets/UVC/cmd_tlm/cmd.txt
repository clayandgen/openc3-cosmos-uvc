###############################################################################
# UVC camera command packets.
#
# Wire format (host -> device, framed by LengthProtocol):
#   [SYNC u16 = 0xAABB][LEN u16 = total bytes][PKT_ID u8][PAYLOAD...]
#
# Almost everything uses PKT_ID 0x01 (SET_CTRL). Payload:
#   [UNIT_ID u8][SELECTOR u8][LENGTH u8][SIGNED u8][VALUE i64]
# Each COMMAND fixes UNIT_ID/SELECTOR/LENGTH/SIGNED defaults so the bridge
# stays a thin proxy: one UVC SET_CUR control transfer per command, no name
# table.
#
# Insta360 vendor controls (XU) are gated by uvc_insta360_enabled.
###############################################################################

# -- Camera Terminal controls (unit 1) ----------------------------------------

COMMAND <%= target_name %> ZOOM BIG_ENDIAN "Set absolute zoom (100=1x, 400=4x)"
  APPEND_PARAMETER    SYNC     16 UINT 0xAABB 0xAABB 0xAABB "Sync pattern"
  APPEND_PARAMETER    LEN      16 UINT MIN MAX 0 "Packet length"
  APPEND_ID_PARAMETER PKT_ID    8 UINT 0x01 0x01 0x01 "SET_CTRL"
  APPEND_PARAMETER    UNIT_ID   8 UINT 1 1 1 "CT"
  APPEND_PARAMETER    SELECTOR  8 UINT 0x0B 0x0B 0x0B "ZOOM_ABSOLUTE"
  APPEND_PARAMETER    LENGTH    8 UINT 2 2 2 "bytes"
  APPEND_PARAMETER    SIGNED    8 UINT 0 0 0 "unsigned"
  APPEND_PARAMETER    VALUE    64 INT 100 400 100 "Zoom level"
  SCREEN <%= target_name %> CONTROL

COMMAND <%= target_name %> PAN_TILT BIG_ENDIAN "Set absolute pan + tilt (combined UVC control)"
  APPEND_PARAMETER    SYNC   16 UINT 0xAABB 0xAABB 0xAABB "Sync pattern"
  APPEND_PARAMETER    LEN    16 UINT MIN MAX 0 "Packet length"
  APPEND_ID_PARAMETER PKT_ID  8 UINT 0x02 0x02 0x02 "SET_PANTILT"
  APPEND_PARAMETER    PAN    32 INT -648000 648000 0 "Pan (1/3600 degree)"
  APPEND_PARAMETER    TILT   32 INT -648000 648000 0 "Tilt (1/3600 degree)"
