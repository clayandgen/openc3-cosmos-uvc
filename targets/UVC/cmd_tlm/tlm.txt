###############################################################################
# UVC camera telemetry packet.
#
# Wire format (device -> host, framed by LengthProtocol):
#   [SYNC u16 = 0xAABB][LEN u16][PKT_ID u8 = 0x80][PAYLOAD]
#
# Bridge does NOT synthesize telemetry. STATUS is built on demand from live
# `v4l2-ctl --get-ctrl` queries against the actual camera; MODEL_ID/MODEL come
# from real USB enumeration. XU vendor state (AI tracking, scene mode, etc.)
# is write-only here — no STATUS field for it until a libusb GET_CUR path is
# wired up.
###############################################################################

TELEMETRY <%= target_name %> STATUS BIG_ENDIAN "Live V4L2 control snapshot"
  APPEND_ITEM    SYNC   16 UINT "Sync pattern"
    FORMAT_STRING "0x%04X"
  APPEND_ITEM    LEN    16 UINT "Packet length"
  APPEND_ID_ITEM PKT_ID 8  UINT 0x80 "Packet ID"

  APPEND_ITEM PAN  32 INT  "Pan position (1/100 deg)"
  APPEND_ITEM TILT 32 INT  "Tilt position (1/100 deg)"
  APPEND_ITEM ZOOM 32 INT  "Zoom (100=1x, 400=4x)"

  APPEND_ITEM BRIGHTNESS 32 INT "Brightness"
  APPEND_ITEM CONTRAST   32 INT "Contrast"
  APPEND_ITEM SATURATION 32 INT "Saturation"
  APPEND_ITEM SHARPNESS  32 INT "Sharpness"
  APPEND_ITEM GAIN       32 INT "Gain"
  APPEND_ITEM BACKLIGHT  32 INT "Backlight compensation"
    STATE OFF 0
    STATE ON  1

  APPEND_ITEM AWB     32 INT "Auto white balance"
    STATE OFF 0
    STATE ON  1
  APPEND_ITEM WB_TEMP 32 INT "White balance temperature"
    UNITS Kelvin K

  APPEND_ITEM EXP_AUTO 32 INT "Exposure mode (V4L2 enum: 1=manual, 3=auto)"
    STATE MANUAL 1
    STATE AUTO   3
  APPEND_ITEM EXP_ABS  32 INT "Exposure absolute"

  APPEND_ITEM AF        32 INT "Auto focus"
    STATE OFF 0
    STATE ON  1
  APPEND_ITEM FOCUS_ABS 32 INT "Focus position"

  APPEND_ITEM MODEL_ID 16 UINT "USB PID (0 if unknown)"
    FORMAT_STRING "0x%04X"
  APPEND_ITEM MODEL 0 STRING "Camera model string (from USB enumeration)"
